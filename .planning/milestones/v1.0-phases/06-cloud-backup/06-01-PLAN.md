Module-scope state (NOT exported):

Exported functions:

isGoogleConnected() → boolean
  Returns true if localStorage.getItem(CLOUD_PROVIDER_KEY) === 'google'

fetchGoogleUserEmail(accessToken) → Promise<string>
  GET https://www.googleapis.com/oauth2/v3/userinfo with Authorization: Bearer header
  Returns response.json().email

initGoogleDrive()
  Creates _tokenClient via google.accounts.oauth2.initTokenClient() if not already created.
  Client config: { client_id: GOOGLE_CLIENT_ID, scope: GOOGLE_SCOPES, callback: ... }
  The callback is set per-call (see withGoogleToken). Use error_callback for popup blocks.
  Guard: if (!window.google) throw new Error('GIS_NOT_LOADED')

withGoogleToken(callback) → void
  Called ONLY from user-initiated click handlers (popup blocker constraint).
  If _tokenData exists AND Date.now() < _tokenData.expires_at → call callback(_tokenData.access_token) directly.
  Otherwise → call _tokenClient.requestAccessToken({ prompt: '' }) with callback wired as:
    callback field: async (response) => {
      if (response.error) throw new Error('GOOGLE_AUTH_FAILED: ' + response.error);
      _tokenData = { access_token: response.access_token, expires_at: Date.now() + (response.expires_in * 1000) - 60_000 };
      const email = await fetchGoogleUserEmail(_tokenData.access_token);
      localStorage.setItem(CLOUD_PROVIDER_KEY, 'google');
      localStorage.setItem(CLOUD_ACCOUNT_KEY, email);
      callback(_tokenData.access_token);
    }
  Important: initGoogleDrive() must be called before withGoogleToken. The callback is replaced each call by creating a fresh token client config each time OR by capturing callback in closure and updating _tokenClient's callback. Use the approach: store callback in module var _pendingCallback, set _tokenClient callback to always call _pendingCallback.

disconnectGoogle()
  _tokenData = null;
  _tokenClient = null;
  localStorage.removeItem(CLOUD_PROVIDER_KEY);
  localStorage.removeItem(CLOUD_ACCOUNT_KEY);
  localStorage.removeItem(CLOUD_LAST_BACKUP_KEY);
  If window.google?.accounts?.oauth2 exists: google.accounts.oauth2.revoke(token, () => {}) — best effort, don't await.

googleDriveUpload(jsonString, accessToken) → Promise<{ id, modifiedTime }>
  Step 1: List existing file — GET https://www.googleapis.com/drive/v3/files?spaces=appDataFolder&q=name%3D%27budget-backup.json%27&fields=files(id) with Auth header
  Step 2: If existing file found, PATCH https://www.googleapis.com/upload/drive/v3/files/{id}?uploadType=multipart
          If no existing file, POST https://www.googleapis.com/upload/drive/v3/files?uploadType=multipart with parents: ['appDataFolder']
  Body: FormData with metadata blob (application/json) + file blob (application/json)
  On success: localStorage.setItem(CLOUD_LAST_BACKUP_KEY, new Date().toISOString())
  Return parsed response JSON. Throw if !res.ok with message 'Drive upload failed: {status}'.

googleDriveDownload(accessToken) → Promise<string>
  Step 1: List file — same query as upload step 1, fields=files(id,modifiedTime)
  If no file: throw new Error('NO_BACKUP_FOUND')
  Step 2: GET https://www.googleapis.com/drive/v3/files/{id}?alt=media with Auth header
  If !res.ok: throw new Error('Drive download failed: {status}')
  Return res.text() — raw JSON string

Constants at top:
  const ONEDRIVE_CLIENT_ID = import.meta.env.VITE_ONEDRIVE_CLIENT_ID ?? '';
  const ONEDRIVE_FILE_NAME = 'budget-backup.json';
  const MSAL_SCOPES = ['Files.ReadWrite', 'User.Read'];
  const CLOUD_PROVIDER_KEY = 'cloud_provider';
  const CLOUD_ACCOUNT_KEY = 'cloud_account_email';
  const CLOUD_LAST_BACKUP_KEY = 'cloud_last_backup';

Module-scope state:
  let _msalInstance = null;
  let _initialized = false;

getMsal() → Promise<PublicClientApplication>
  Creates singleton if not exists:
    new PublicClientApplication({
      auth: { clientId: ONEDRIVE_CLIENT_ID, authority: 'https://login.microsoftonline.com/consumers', redirectUri: window.location.origin },
      cache: { cacheLocation: 'localStorage', storeAuthStateInCookie: false }
    })
  If !_initialized: await _msalInstance.initialize(); _initialized = true;
  Return _msalInstance;

getOneDriveToken() → Promise<string>
  const msal = await getMsal();
  const accounts = msal.getAllAccounts();
  if (!accounts.length) throw new Error('NOT_CONNECTED');
  try { const res = await msal.acquireTokenSilent({ scopes: MSAL_SCOPES, account: accounts[0] }); return res.accessToken; }
  catch (err) { if (err instanceof InteractionRequiredAuthError) { const res = await msal.acquireTokenPopup({ scopes: MSAL_SCOPES, account: accounts[0] }); return res.accessToken; } throw err; }

connectOneDrive() → Promise<{ email: string }>
  const msal = await getMsal();
  const response = await msal.loginPopup({ scopes: MSAL_SCOPES });
  const email = response.account?.username ?? '';
  localStorage.setItem(CLOUD_PROVIDER_KEY, 'onedrive');
  localStorage.setItem(CLOUD_ACCOUNT_KEY, email);
  return { email };

disconnectOneDrive() → Promise<void>
  const msal = await getMsal();
  const accounts = msal.getAllAccounts();
  if (accounts.length) { await msal.clearCache(); } // clears msal.* localStorage keys
  _msalInstance = null; _initialized = false;
  localStorage.removeItem(CLOUD_PROVIDER_KEY);
  localStorage.removeItem(CLOUD_ACCOUNT_KEY);
  localStorage.removeItem(CLOUD_LAST_BACKUP_KEY);

isOneDriveConnected() → boolean
  Returns localStorage.getItem(CLOUD_PROVIDER_KEY) === 'onedrive'

oneDriveUpload(jsonString) → Promise<{ id, lastModifiedDateTime }>
  const token = await getOneDriveToken();
  PUT https://graph.microsoft.com/v1.0/me/drive/root:/{ONEDRIVE_FILE_NAME}:/content
  Headers: { Authorization: 'Bearer ' + token, 'Content-Type': 'application/json' }
  Body: jsonString
  If !res.ok: throw new Error('OneDrive upload failed: ' + res.status)
  On success: localStorage.setItem(CLOUD_LAST_BACKUP_KEY, new Date().toISOString())
  Return parsed JSON.

oneDriveDownload() → Promise<string>
  const token = await getOneDriveToken();
  GET https://graph.microsoft.com/v1.0/me/drive/root:/{ONEDRIVE_FILE_NAME}:/content
  If res.status === 404: throw new Error('NO_BACKUP_FOUND')
  If !res.ok: throw new Error('OneDrive download failed: ' + res.status)
  Return res.text()

getOneDriveUserEmail() → Promise<string>
  const token = await getOneDriveToken();
  GET https://graph.microsoft.com/v1.0/me?$select=mail,userPrincipalName with Auth header
  Return data.mail || data.userPrincipalName

Import at top: import { PublicClientApplication, InteractionRequiredAuthError } from '@azure/msal-browser';
